5: Explorar parte 2: extraer y agrupar

Explorar contenido a partir de extraer ciertos datos y de agruparlos con respecto a un conjunto de categorías de interés.

xplore2_FAOSTAT.R 
                
                # Exploraremos la categoría RFN que tenemos en rfn_data pero, para un bloque de países, 
                # ...la informaación extraída la guardaremos en el objeto "BLOQUE" 
                BLOQUE <- rfn_data |> filter(area %in% c("Argentina","Brazil","Colombia", "Mexico"))
                
                # .....
                str(BLOQUE)
                
                # Elegiremos 3 registros de BLOQUE para mostrar los valores observados en todas sus variables
                BLOQUE[c(1,1750,3139), ]
                
                # Mostraremos los valores únicos que contiene la variable "item"
                unique(BLOQUE$item)
                
                # Mostraremos los valores únicos que contiene la variable "element"
                unique(BLOQUE$element)
                
                
                # los campos "yeaar_code" y "year" puede contener años individuales, pero en algunos casos
                # se encontrará información por periodo, digamos 2001-2004.
                # Filtraremos solo aquella información donde, ya sea el campo "year_code" o el campo "year" contenga
                # cuatro caracteres, lo que asegura que es información de un año específico y 
                # no de un período. Los datos filtrados los guardaremos en un objeto llamado BLOQUE1
                
                BLOQUE1 <- BLOQUE %>% filter(grepl("^\\d{4}$", year_code))
                
                # Ahora actualizaremos la info de BLOQUE1 filtrando la información de "fertilizantes por nutriente" de 
                # ...acuerdo a su "uso por superficie de cultivo" determinado por el campo "element"
                BLOQUE1 <- BLOQUE1 |> filter(element =="use_per_area_of_cropland")
                
                
                # veamos cómo se disribuye la variable "value" que en unidades 
                # dadas por el campo "unit", en este caso kg/ha (kilogramos por hectárea)
                # para cualquiera de los tres nutrientes posibles en cualquiera de los píses
                # de BLOQUE1 para cualquier año contenido en este objeto
                
                hist(BLOQUE1$value)
                

                
                
                # analizaremos nuestra información contenida en BLOQUE1 pero
                # esta vez agruparemos con respecto a país y nutriente (fertilizante)
                # y generaremos media, desviación estándar y error estándar con respecto a "value"
                # los resultados se guardarán en un nuevo objeto llamado RFN_grouped
                
                
                RFN_grouped <- BLOQUE1 %>% 
                  group_by(area, item) %>%
                  summarise(
                    Mean = mean(value, na.rm = TRUE),
                    SD = sd(value, na.rm = TRUE),
                    SE = SD / sqrt(n()),
                    .groups = "drop"
                  ) %>% 
                  filter(!is.na(Mean))
                  
                # mostremos lo que obtuvimos
               RFN_grouped
               
               # veamos el comportamiento de fertilizantes por superficie cultivada en los 90's
               RFN_90s <- BLOQUE1 %>%
                  filter(year %in% c(1990:1999)) %>%  # Filtra los años deseados
                  group_by(area, item) %>%
                  summarise(
                    Mean = mean(value, na.rm = TRUE),
                    SD = sd(value, na.rm = TRUE),
                    SE = SD / sqrt(n()),
                    .groups = "drop"
                  ) %>% 
                  filter(!is.na(Mean))
                
                 # mostremos lo que obtuvimos
                RFN_90s
                
                 # siguiente década
                RFN_00s <- BLOQUE1 %>%
                  filter(year %in% c(2000:2009)) %>%  # Filtra los años deseados
                  group_by(area, item) %>%
                  summarise(
                    Mean = mean(value, na.rm = TRUE),
                    SD = sd(value, na.rm = TRUE),
                    SE = SD / sqrt(n()),
                    .groups = "drop"
                  ) %>% 
                  filter(!is.na(Mean))
                
                 # mostremos lo que obtuvimos
                RFN_00s
                
                
                 # siguiente década
                RFN_10s <- BLOQUE1 %>%
                  filter(year %in% c(2010:2019)) %>%  # Filtra los años deseados
                  group_by(area, item) %>%
                  summarise(
                    Mean = mean(value, na.rm = TRUE),
                    SD = sd(value, na.rm = TRUE),
                    SE = SD / sqrt(n()),
                    .groups = "drop"
                  ) %>% 
                  filter(!is.na(Mean))
                
                 # mostremos lo que obtuvimos
                RFN_10s
                
                 
                # Grafiquemos esto último

                # los paquetes para graficar y exportar a html son los siguientes pero,
                # los presento como comentarios porque ya fueron cargados desde un inicio

                #library(ggplot2)
                #library(plotly)
                #library(htmlwidgets) 

                # 1. Crear el gráfico base con ggplot2 y guardarlo en una variable estática
                
                
                p_gg <- ggplot(RFN_10s, aes(x = area, y = Mean, fill = item)) +
                  geom_bar(stat = "identity", position = position_dodge(width = 0.8), width = 0.7) +
                  geom_errorbar(
                    aes(ymin = Mean - SE, ymax = Mean + SE), 
                    position = position_dodge(width = 0.8), 
                    width = 0.25
                  ) +
                  labs(
                    title = "Uso promedio de fertilizantes por nutriente por país (2010-2019)",
                    x = "País / Área",
                    y = "Media y Error Estándar (SE)",
                    fill = "Nutriente"
                  ) +
                  theme_minimal()
                  
                  # mostremos lo que obtuvimos
                 p_gg
                 
                 
                 
                # 2. Exportar como PNG (usando el objeto estático p_gg)
                ggsave(
                  filename = "grafico_nutrientes.png", # al no definir la ruta se guarda en la carpeta donde está este proyecto R
                  plot = p_gg,
                  width = 20,    # Ancho en centímetros
                  height = 12,   # Alto en centímetros
                  units = "cm",  # Unidad de medida
                  dpi = 300      # Resolución en puntos por pulgada
                )
                
                
                # 4. Crear la versión interactiva para HTML
                p_plotly <- ggplotly(p_gg)
                
                 # mostremos lo que obtuvimos
                p_plotly
                
                # 5. Guardar como archivo HTML
                saveWidget(
                  widget = p_plotly, 
                  file = "grafico_nutrientes_2010s.html", 
                  selfcontained = TRUE
                )
